CREATE DATABASE IF NOT EXISTS jurnalisti;

CREATE TABLE IF NOT EXISTS user (
id INT(6) NOT NULL AUTO_INCREMENT,
nume VARCHAR(30) NOT NULL,
prenume VARCHAR(30) NOT NULL,
username VARCHAR(40) NOT NULL,
parola VARCHAR(30) NOT NULL,
email VARCHAR(30) NOT NULL,
id_rol VARCHAR(5) NOT NULL,
id_categorie INT(4),
PRIMARY KEY (id)
);

ALTER TABLE user ADD CONSTRAINT user_email_format CHECK (email
LIKE '%@%.%');


CREATE TABLE IF NOT EXISTS categorie (
id_categorie INT(4) NOT NULL,
nume_categorie VARCHAR(30) NOT NULL,
PRIMARY KEY (id_categorie)
);


CREATE TABLE IF NOT EXISTS rol (
id_rol VARCHAR(5) NOT NULL,
denumire_rol VARCHAR(30) NOT NULL,
PRIMARY KEY (id_rol)
);

CREATE TABLE IF NOT EXISTS articol (
id_articol INT(5) NOT NULL,
titlu VARCHAR(100) NOT NULL,
data_creare DATETIME NOT NULL,
data_modificare DATETIME,
continut_articol VARCHAR(500) NOT NULL,
autor INT(6) NOT NULL,
editor INT(6) NOT NULL,
status VARCHAR(30) NOT NULL,
id_categorie INT(4) NOT NULL,
PRIMARY KEY (id_articol)
);

CREATE TABLE IF NOT EXISTS comentariu (
id_comentariu VARCHAR(5) NOT NULL,
titlu VARCHAR(100) NOT NULL,
continut_comentariu VARCHAR(300) NOT NULL,
data_adaugarii DATETIME NOT NULL,
id_articol INT(5) NOT NULL,
id_user INT(6) NOT NULL,
PRIMARY KEY (id_comentariu)
);
